### ~={green}TanStack Query =~
####  ~={blue}where to use=~ 
- Client-side API fetching
- Dashboards
- Infinite Scroll
- Pagination 
- Frequently updated data
- Instead of writing repetitive fetching logic, we use query hooks like `useQuery()`
---
#### ~={1}QueryClientProvider=~
- To make TanStack Query available across the app, we wrap the application with QueryClientProvider.
- This provider gives all components access to the query client and cache system.  

---
#### Core TanStack Query Terms
 1. ~={3}`Query`=~: fetch data from s server
 2. ~={3}`Mutation`=~: Change data on the server. -> POST/PUT/PATCH/DELETE
 3. ~={3}`Cache`=~: TanStack Query stores fetched data in a cache so the same request doesn't to be made again.
 4. ~={3}`queryKey`=~: A query key uniquely identifies a query in the cache. 
 5. ~={3}`Stale Data`=~: No longer be up-to-date. 
 6. ~={3}`refetch`:=~ Fetching the data again from the server. 
 7. ~={3}`Invalidation`=~: I want to update Stale data. 
 8. ~={3}`Query Client`=~: The query client manage the cache and all queries.

```tsx

//                providers/query-client-provider.tsx
"use client";

import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { useState } from "react";

type Props = {
    children: React.ReactNode;
};
  
const QueryProvider = ({children}: Props) => {  
    const [queryClient] = useState(() => new QueryClient());

    return (
        <QueryClientProvider client={queryClient}>
            {children}
        </QueryClientProvider>
    )
}

export default QueryProvider
```

### ~={orange}useQuery=~
- ~={3}`useQuery`=~ is the primary hook in **TanStack Query** for fetching, caching, and synchronizing server data.

#### Without `useQuery` (Traditional React)
```tsx
"use client";

import { useEffect, useState } from "react";

interface User {
  id: number;
  name: string;
  email: string;
}

export default function Users() {
  const [users, setUsers] = useState<User[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    async function fetchUsers() {
      try {
        const res = await fetch("https://jsonplaceholder.typicode.com/users");

        if (!res.ok) {
          throw new Error("Failed to fetch users");
        }

        const data: User[] = await res.json();
        setUsers(data);
      } catch (err) {
        if (err instanceof Error) {
          setError(err.message);
        }
      } finally {
        setLoading(false);
      }
    }

    fetchUsers();
  }, []);

  if (loading) return <p>Loading...</p>;
  if (error) return <p>{error}</p>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

### With `useQuery`

```tsx
"use client";

import { useQuery } from "@tanstack/react-query";

interface User {
  id: number;
  name: string;
  email: string;
}

async function fetchUsers(): Promise<User[]> {
  const res = await fetch("https://jsonplaceholder.typicode.com/users");

  if (!res.ok) {
    throw new Error("Failed to fetch users");
  }

  return res.json();
}

export default function Users() {
  const {
    data: users,
    isLoading,
    error,
  } = useQuery<User[]>({
    queryKey: ["users"],
    queryFn: fetchUsers,
  });

  if (isLoading) return <p>Loading...</p>;

  if (error instanceof Error) {
    return <p>{error.message}</p>;
  }

  return (
    <ul>
      {users?.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```
