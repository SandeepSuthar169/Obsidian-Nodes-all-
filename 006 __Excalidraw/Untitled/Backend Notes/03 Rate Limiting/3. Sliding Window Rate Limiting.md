# Sliding Window Rate Limiting
- The Sliding window algorithm limits how many requests a client can make during a rolling time period.

```ts

import type {
  Request,
  Response,
  NextFunction,
} from "express";

type RateLimiterOptions = {
  windowMs?: number;
  maxRequests?: number;
};

export function slidingWindowRateLimiter({
  windowMs = 60_000,
  maxRequests = 5,
}: RateLimiterOptions = {}) {
  // IP address -> timestamps of accepted requests
  const requests = new Map<string, number[]>();

  // Periodically remove stale client entries to control memory use
  const cleanupTimer = setInterval(() => {
    const cutoff = Date.now() - windowMs;

    for (const [ip, timestamps] of requests) {
      const activeTimestamps = timestamps.filter(
        (timestamp) => timestamp > cutoff
      );

      if (activeTimestamps.length === 0) {
        requests.delete(ip);
      } else {
        requests.set(ip, activeTimestamps);
      }
    }
  }, windowMs);

  cleanupTimer.unref?.();

  return (
    req: Request,
    res: Response,
    next: NextFunction
  ) => {
    const ip = req.ip ?? "unknown";
    const now = Date.now();
    const windowStart = now - windowMs;

    // Keep only requests from the most recent time window
    const timestamps = (requests.get(ip) ?? []).filter(
      (timestamp) => timestamp > windowStart
    );

    res.setHeader("X-RateLimit-Limit", maxRequests);

    // Reject if the client has already used its quota
    if (timestamps.length >= maxRequests) {
      requests.set(ip, timestamps);

      const retryAfterMs =
        timestamps[0] + windowMs - now;

      res.setHeader("X-RateLimit-Remaining", 0);
      res.setHeader(
        "Retry-After",
        Math.max(1, Math.ceil(retryAfterMs / 1000))
      );

      return res.status(429).json({
        error: "Too many requests",
        message: "Please try again later.",
      });
    }

    // Record this accepted request
    timestamps.push(now);
    requests.set(ip, timestamps);

    res.setHeader(
      "X-RateLimit-Remaining",
      maxRequests - timestamps.length
    );

    next();
  };
}
```


```ts

import express from "express";
import { slidingWindowRateLimiter } from "./rateLimiter";

const app = express();

const limiter = slidingWindowRateLimiter({
  windowMs: 60 * 1000, // 60 seconds
  maxRequests: 5,      // 5 requests per sliding window
});

app.use("/api", limiter);

app.get("/api/users", (_req, res) => {
  res.json({
    message: "Users fetched successfully",
  });
});

app.listen(3000, () => {
  console.log("Server running on port 3000");
});
```
