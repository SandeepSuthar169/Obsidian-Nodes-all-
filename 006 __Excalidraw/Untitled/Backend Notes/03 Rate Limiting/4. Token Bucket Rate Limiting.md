# Token Bucket Rate Limiting
- Controls how frequently clients can make requests while still allowing short bursts of traffic. 
- Controls how many requests a user can make within a certain period of time.

![[OIP-1246077152.jpg]]

Suppose these are your settings:

| Setting          | Value          |
| ---------------- | -------------- |
| Bucket capacity  | 5 tokens       |
| Refill rate      | 1 token/second |
| Cost per request | 1 token        |

Imagine you send 5 requests immediately.
### Remember these three things
- Capacity: The maximum number of tokens the bucket can hold.
- Refill rate: How quickly tokens become available again.
- Request cost: How many tokens a request consumes.

---
```ts
//                   src/middleware/tokenBucket.ts.
import type { Request, Response, NextFunction } from "express";

interface Bucket {
  tokens: number;
  lastRefill: number;
}

interface TokenBucketOptions {
  capacity: number;
  refillRate: number; // Tokens per second
  requestCost?: number;
}

export function tokenBucketRateLimit({
  capacity,
  refillRate,
  requestCost = 1,
}: TokenBucketOptions) {
  if (
    !Number.isFinite(capacity) ||
    capacity <= 0 ||
    !Number.isFinite(refillRate) ||
    refillRate <= 0 ||
    !Number.isFinite(requestCost) ||
    requestCost <= 0 ||
    requestCost > capacity
  ) {
    throw new Error("Invalid token bucket configuration");
  }

  // In-memory storage: one bucket per client IP.
  const buckets = new Map<string, Bucket>();

  return (
    req: Request,
    res: Response,
    next: NextFunction
  ): void => {
    const key = req.ip ?? req.socket.remoteAddress ?? "unknown";
    const now = Date.now();

    let bucket = buckets.get(key);

    // First request: initialize a full bucket.
    if (!bucket) {
      bucket = {
        tokens: capacity,
        lastRefill: now,
      };
      buckets.set(key, bucket);
    }

    // 1. Calculate elapsed time.
    const elapsedSeconds = (now - bucket.lastRefill) / 1000;

    // 2. Refill tokens, but never exceed capacity.
    bucket.tokens = Math.min(
      capacity,
      bucket.tokens + elapsedSeconds * refillRate
    );

    bucket.lastRefill = now;

    // 3. Reject if there aren't enough tokens.
    if (bucket.tokens < requestCost) {
      const secondsToWait =
        (requestCost - bucket.tokens) / refillRate;

      res.setHeader("Retry-After", Math.ceil(secondsToWait));
      res.setHeader("X-RateLimit-Limit", capacity);
      res.setHeader("X-RateLimit-Remaining", 0);

      res.status(429).json({
        error: "Too many requests",
        message: "Please try again later.",
        retryAfterSeconds: Math.ceil(secondsToWait),
      });

      return;
    }

    // 4. Consume tokens and allow the request.
    bucket.tokens -= requestCost;

    res.setHeader("X-RateLimit-Limit", capacity);
    res.setHeader(
      "X-RateLimit-Remaining",
      Math.floor(bucket.tokens)
    );

    next();
  };
}

```

```ts
//                              src/server.ts.
import express from "express";
import { tokenBucketRateLimit } from "./middleware/tokenBucket";

const app = express();

app.use(express.json());

// Allow 5 requests initially.
// Refill 1 token every second.
const rateLimiter = tokenBucketRateLimit({
  capacity: 5,
  refillRate: 1,
});

app.get("/api/data", rateLimiter, (req, res) => {
  res.json({
    message: "Request successful!",
    data: {
      id: 1,
      name: "Example data",
    },
  });
});

app.get("/", (req, res) => {
  res.send("Token Bucket Rate Limiter is running!");
});

app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});
```
