let count = 0

for(let  i = 0; i < arr.length - 2; i++){
	let left = i + 1
	let right = arr.length - 1
	
	while (left < right){
		sum =  arr[i] + arrp[left] + arr[right] 
		
		if(sum < target){
			count += right - left
			left++
		} else {
			right--
		}
	
	}
}
return count
