---
- Fix one number and use **two pointers** to find the other two numbers.

### 1.  Sort the array
```js
nums.sort((a, b) => a - b);
```

### 2.  Fix one element
Use `i` as the first number.
```txt
i
↓
[-4, -1, -1, 0, 1, 2]
```

### 3. Use two pointers
```js
left = i + 1
right = nums.length - 1
```

```js
sum = nums[i] + nums[left] + nums[right];
```
- If `sum === 0` → found a triplet
- If `sum < 0` → move `left++`
- If `sum > 0` → move `right--`

### 4. Skip duplicate fixed elements:

```js
if (i > 0 && nums[i] === nums[i - 1]) continue;
```
After finding a triplets
```js
while (left < right && nums[left] === nums[left + 1]) {
    left++;
}

while (left < right && nums[right] === nums[right - 1]) {
    right--;
}
```

```js
function threeSum(nums) {
    const result = [];

    nums.sort((a, b) => a - b);

    for (let i = 0; i < nums.length - 2; i++) {

        // Skip duplicates
        if (i > 0 && nums[i] === nums[i - 1]) {
            continue;
        }

        let left = i + 1;
        let right = nums.length - 1;

        while (left < right) {
            const sum = nums[i] + nums[left] + nums[right];

            if (sum === 0) {
                result.push([nums[i], nums[left], nums[right]]);

                // Skip duplicates
                while (left < right && nums[left] === nums[left + 1]) {
                    left++;
                }

                while (left < right && nums[right] === nums[right - 1]) {
                    right--;
                }

                left++;
                right--;
            }
            else if (sum < 0) {
                left++;
            }
            else {
                right--;
            }
        }
    }

    return result;
}
```
